  bot.callbackQuery('settings:back', async (ctx) => {
    try {
      const { settingsCommand } = await import('../commands/settings.js');
      await ctx.answerCallbackQuery();
      await settingsCommand(ctx);
    } catch (e) {
      console.error('Settings back error:', e);
      await ctx.answerCallbackQuery({ text: '❌ ' + e.message, show_alert: true });
    }
  });
