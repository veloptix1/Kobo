const API = '/admin/api/admin/data';

async function apiGet(resource) {
  const r = await fetch(API + '?resource=' + resource);
  return r.json();
}
async function apiSend(resource, method, body) {
  const r = await fetch(API + '?resource=' + resource, {
    method, headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
  return r.json();
}

function escapeHtml(s) { return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
function emptyBox(msg) { return '<div class="empty">' + msg + '</div>'; }
function timeAgo(iso) {
  if (!iso) return '—';
  const diff = Math.floor((Date.now() - new Date(iso).getTime()) / 1000);
  if (diff < 60) return 'À l\'instant';
  if (diff < 3600) return 'Il y a ' + Math.floor(diff/60) + ' min';
  if (diff < 86400) return 'Il y a ' + Math.floor(diff/3600) + ' h';
  if (diff < 604800) return 'Il y a ' + Math.floor(diff/86400) + ' j';
  return new Date(iso).toLocaleDateString('fr-FR');
}
function showToast(msg, isError) {
  const t = document.getElementById('toast');
  document.getElementById('toastText').textContent = msg;
  t.className = 'toast show' + (isError ? ' error' : '');
  setTimeout(() => t.classList.remove('show'), 3000);
}
function openModal(title, html) {
  document.getElementById('modalTitle').textContent = title;
  document.getElementById('modalContent').innerHTML = html;
  document.getElementById('modalOverlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function closeModal() { document.getElementById('modalOverlay').classList.remove('open'); document.body.style.overflow = ''; }
function closeModalOutside(e) { if (e.target.id === 'modalOverlay') closeModal(); }

(async function init() {
  const loader = document.getElementById('loader');
  try {
    const [retraits, tasks, users, settings] = await Promise.all([
      apiGet('withdrawals'),
      apiGet('tasks'),
      apiGet('users'),
      apiGet('settings'),
    ]);
    DATA.retraits = retraits.data || [];
    DATA.tasks = tasks.data || [];
    DATA.users = users.data || [];
    DATA.settings = settings.data || [];
    DATA.admins = DATA.users.filter(u => u.is_admin);

    try { await loadStats(); } catch (e) { console.error('loadStats', e); }
    try { renderRetraits(); } catch (e) { console.error('renderRetraits', e); }
    try { renderTasks(); } catch (e) { console.error('renderTasks', e); }
    try { renderUsers(); } catch (e) { console.error('renderUsers', e); }
    try { renderAdmins(); } catch (e) { console.error('renderAdmins', e); }
    try { renderParrainage(); } catch (e) { console.error('renderParrainage', e); }
    try { renderSettings(); } catch (e) { console.error('renderSettings', e); }
    try { loadAdvertisers(); } catch (e) { console.error('loadAdvertisers', e); }

    setTimeout(() => { if (loader) loader.classList.add('hidden'); }, 300);
  } catch (e) {
    console.error(e);
    if (loader) loader.innerHTML = '<div style="color:white;padding:20px;font-size:12px;font-family:monospace;">❌ Erreur init:<br>' + e.message + '</div>';
  }
})();

async function loadStats() {
  const grid = document.getElementById('statsGrid');
  if (!grid) return;
  grid.innerHTML = '<div class="stat-box"><div class="lbl">Chargement...</div><div class="val">—</div></div>';
  const s = await apiGet('stats');
  const stats = [
    { l: 'Utilisateurs', v: s.totalUsers || 0 },
    { l: 'Bannis', v: s.bannedUsers || 0 },
    { l: 'Retraits en attente', v: s.pendingWD || 0 },
    { l: 'Kobo en circulation', v: Math.round(s.totalKobo || 0) },
    { l: 'Total distribué', v: Math.round(s.totalEarned || 0) },
    { l: 'Total retiré', v: Math.round(s.totalWithdrawn || 0) },
  ];
  grid.innerHTML = stats.map(x => '<div class="stat-box"><div class="lbl">' + x.l + '</div><div class="val">' + x.v + '</div></div>').join('');
  const badge = document.getElementById('retraitsBadge');
  if (badge) {
    if (s.pendingWD > 0) { badge.textContent = s.pendingWD; badge.classList.add('show'); }
    else badge.classList.remove('show');
  }
}

async function wdAction(id, status) {
  if (!confirm('Confirmer ' + (status === 'paid' ? 'le paiement' : 'le refus') + ' ?')) return;
  const r = await apiSend('withdrawal', 'PATCH', { id, status });
  if (r.error) return showToast('Erreur : ' + r.error, true);
  showToast('✅ Retrait mis à jour');
  const fresh = await apiGet('withdrawals');
  DATA.retraits = fresh.data || [];
  renderRetraits(); loadStats();
}

async function saveTask(id) {
  const advSelect = document.getElementById('tAdvertiser');
  const payload = {
    title: document.getElementById('tTitle').value.trim(),
    description: document.getElementById('tDesc').value.trim() || null,
    type: document.getElementById('tType').value,
    reward: Number(document.getElementById('tReward').value),
    advertiser_id: advSelect && advSelect.value ? Number(advSelect.value) : null,
    cost_per_completion: Number(document.getElementById('tCost').value) || 0,
    link: document.getElementById('tLink').value.trim() || null,
    channel_id: document.getElementById('tChannel').value.trim() || null,
    daily_limit: Number(document.getElementById('tDaily').value),
    is_active: document.getElementById('tActive').checked,
  };
  if (!payload.title) return showToast('Titre requis', true);
  const r = id ? await apiSend('task', 'PATCH', Object.assign({ id }, payload)) : await apiSend('task', 'POST', payload);
  if (r.error) return showToast('Erreur : ' + r.error, true);
  showToast(id ? 'Modifié' : 'Créée');
  closeModal();
  const tasks = await apiGet('tasks'); DATA.tasks = tasks.data || [];
  renderTasks();
}

async function toggleTask(id) {
  const t = DATA.tasks.find(x => x.id === id); if (!t) return;
  await apiSend('task', 'PATCH', { id, is_active: !t.is_active });
  const tasks = await apiGet('tasks'); DATA.tasks = tasks.data || [];
  renderTasks();
}

async function deleteTask(id) {
  if (!confirm('Supprimer ?')) return;
  await apiSend('task', 'DELETE', { id });
  showToast('Supprimée');
  const tasks = await apiGet('tasks'); DATA.tasks = tasks.data || [];
  renderTasks();
}

function renderParrainage() {
  const list = document.getElementById('parrainList');
  const stats = document.getElementById('parrainStats');
  if (!list || !stats) return;
  const groups = {};
  DATA.users.forEach(u => {
    if (!u.referred_by) return;
    if (!groups[u.referred_by]) {
      const parrain = DATA.users.find(x => x.telegram_id === u.referred_by);
      groups[u.referred_by] = { parrain: parrain || { telegram_id: u.referred_by, first_name: 'Inconnu' }, filleuls: [] };
    }
    groups[u.referred_by].filleuls.push(u);
  });
  const arr = Object.values(groups).sort((a, b) => b.filleuls.length - a.filleuls.length);
  stats.innerHTML = '<div class="stat-box"><div class="lbl">Parrains actifs</div><div class="val">' + arr.length + '</div></div>' +
    '<div class="stat-box" style="border-color:var(--kobo-orange);"><div class="lbl">Total filleuls</div><div class="val">' + DATA.users.filter(u => u.referred_by).length + '</div></div>';
  const q = (document.getElementById('parrainSearch')?.value || '').toLowerCase().trim();
  let f = arr;
  if (q) f = f.filter(g => (g.parrain.first_name||'').toLowerCase().includes(q) || (g.parrain.username||'').toLowerCase().includes(q));
  if (f.length === 0) { list.innerHTML = emptyBox('Aucun parrain'); return; }
  list.innerHTML = f.slice(0, 50).map(g => {
    const p = g.parrain; const c = g.filleuls.length;

function renderSettings() {
  const list = document.getElementById('settingsList');
  if (!list) return;
  const q = (document.getElementById('settingsSearch')?.value || '').toLowerCase().trim();
  let f = DATA.settings;
  if (q) f = f.filter(s => s.key.toLowerCase().includes(q) || (s.description||'').toLowerCase().includes(q));
  if (f.length === 0) { list.innerHTML = emptyBox('Aucun paramètre'); return; }

  const BOOLEAN_KEYS = ['mining_enabled','wheel_enabled','slots_enabled','leaderboard_enabled','ads_enabled','vip_enabled','marketplace_enabled','marketplace_user_enabled','vip_priority_withdraw','task_auto_approve','bot_enabled','convert_enabled','recharge_enabled','marketplace_auto_approve','antifraud_enabled','antifraud_block_same_device','antifraud_block_same_ip','antifraud_require_channel','marketplace_allow_real_payment','marketplace_allow_kobo_payment','marketplace_require_moderation'];
  const NUM_ARR_KEYS = ['wheel_prizes','wheel_probabilities','slots_multipliers','slots_probabilities'];
  const METHODS_KEYS = ['withdraw_methods','recharge_methods'];
  const CHANNEL_KEYS = ['withdrawal_channel_id'];
  const CHANNELS_ARR = ['mandatory_channels','marketplace_broadcast_channels'];

  list.innerHTML = f.map(s => {
    const key = s.key;
    const desc = escapeHtml(s.description || key);

    if (BOOLEAN_KEYS.includes(key)) {
      const v = s.value === true || s.value === 'true';
      return '<div class="config-row"><div class="cr-info"><div class="cr-label">' + desc + '</div><div class="cr-cle">' + key + '</div></div>' +
        '<label class="toggle"><input type="checkbox" ' + (v ? 'checked' : '') + ' onchange="updateSettingValue(\'' + key + '\', this.checked)"><span class="toggle-slider"></span></label></div>';
    }

    if (METHODS_KEYS.includes(key)) {
      const arr = Array.isArray(s.value) ? s.value : [];
      const methods = [{ v: 'orange_money', l: '🟠 Orange Money' }, { v: 'mtn', l: '🟡 MTN' }, { v: 'wave', l: '🌊 Wave' }, { v: 'usdt', l: '🪙 USDT' }];

        '<div class="checkbox-row">' + methods.map(m => '<label class="checkbox-item ' + (arr.includes(m.v) ? 'checked' : '') + '"><input type="checkbox" ' + (arr.includes(m.v) ? 'checked' : '') + ' onchange="toggleArrayItem(\'' + key + '\', \'' + m.v + '\', this.checked)"> ' + m.l + '</label>').join('') + '</div></div>';
    }

    if (CHANNEL_KEYS.includes(key)) {
      const channels = ['@koboofficial', '@koboretrai', '@kobotravail'];
      const v = String(s.value || '').replace(/"/g, '');
      return '<div class="config-row" style="flex-direction:column;align-items:stretch;"><div class="cr-info" style="margin-bottom:10px;"><div class="cr-label">' + desc + '</div><div class="cr-cle">' + key + '</div></div>' +
        '<select onchange="updateSettingValue(\'' + key + '\', this.value)" style="width:100%;padding:10px;border:1.5px solid var(--border);border-radius:10px;font-size:14px;font-family:inherit;background:#f7faf8;">' +
        channels.map(c => '<option value="' + c + '" ' + (v === c ? 'selected' : '') + '>' + c + '</option>').join('') + '</select></div>';
    }

    if (CHANNELS_ARR.includes(key)) {
      const arr = Array.isArray(s.value) ? s.value : [];
      return '<div class="config-row" style="flex-direction:column;align-items:stretch;"><div class="cr-info" style="margin-bottom:10px;"><div class="cr-label">' + desc + '</div><div class="cr-cle">' + key + '</div></div>' +
        '<div class="tag-list">' + arr.map((item, i) => {
          const name = typeof item === 'object' ? (item.name || item.id) : item;
          return '<span class="tag-item">' + escapeHtml(name) + ' <button type="button" onclick="removeArrayIndex(\'' + key + '\', ' + i + ')">✕</button></span>';
        }).join('') + '<button type="button" class="tag-add" onclick="addArrayChannel(\'' + key + '\')">➕ Ajouter</button></div></div>';
    }

    if (NUM_ARR_KEYS.includes(key)) {
      const arr = Array.isArray(s.value) ? s.value : [];
      return '<div class="config-row" style="flex-direction:column;align-items:stretch;"><div class="cr-info" style="margin-bottom:10px;"><div class="cr-label">' + desc + '</div><div class="cr-cle">' + key + '</div></div>' +

  }).join('');

async function updateSettingValue(key, value) {
  const r = await apiSend('setting', 'PATCH', { key, value });
  if (r.error) return showToast('Erreur : ' + r.error, true);
  const idx = DATA.settings.findIndex(s => s.key === key);
  if (idx >= 0) DATA.settings[idx].value = value;
  showToast('✅ Enregistré');
}

async function toggleBoolSetting(key, checked) { await updateSettingValue(key, checked); }
async function updateSettingSelect(key, value) { await updateSettingValue(key, value); }

async function toggleArrayItem(key, item, checked) {
  const setting = DATA.settings.find(s => s.key === key);
  if (!setting) return;
  let arr = Array.isArray(setting.value) ? setting.value.slice() : [];
  if (checked && !arr.includes(item)) arr.push(item);
  if (!checked) arr = arr.filter(x => x !== item);
  await updateSettingValue(key, arr);
  renderSettings();
}

async function addArrayNumber(key) {
  const v = prompt('Entrer un nombre :');
  if (!v) return;
  const setting = DATA.settings.find(s => s.key === key);
  const arr = Array.isArray(setting.value) ? setting.value.slice() : [];
  arr.push(Number(v));
  await updateSettingValue(key, arr);
  renderSettings();
}

async function updateArrayIndex(key, index, newVal) {
  const setting = DATA.settings.find(s => s.key === key);
  const arr = Array.isArray(setting.value) ? setting.value.slice() : [];
  arr[index] = Number(newVal);
  await updateSettingValue(key, arr);
}

async function removeArrayIndex(key, index) {
  const setting = DATA.settings.find(s => s.key === key);
  const arr = Array.isArray(setting.value) ? setting.value.slice() : [];
  arr.splice(index, 1);
  await updateSettingValue(key, arr);
  renderSettings();
}

async function addArrayChannel(key) {
  const name = prompt('Nom du canal :');
  if (!name) return;
  const id = prompt('ID du canal (ex: @koboofficial) :');
  if (!id) return;
  const setting = DATA.settings.find(s => s.key === key);
  const arr = Array.isArray(setting.value) ? setting.value.slice() : [];
  arr.push({ id, name, url: 'https://t.me/' + id.replace('@', '') });
  await updateSettingValue(key, arr);
  renderSettings();
}

async function sendBroadcast() {
  const msg = document.getElementById('broadcastMsg').value.trim();
  if (!msg) return showToast('Message vide', true);
  const target = document.getElementById('broadcastTarget').value;
  const status = document.getElementById('broadcastStatus');
  let channelId = null;
  if (target === 'channel') {
    const sel = document.getElementById('broadcastChannel').value;
    channelId = sel === 'custom' ? document.getElementById('broadcastChannelCustom').value.trim() : sel;
    if (!channelId) return showToast('Canal requis', true);
  }
  const label = target === 'channel' ? 'le canal ' + channelId : 'TOUS les utilisateurs';
  if (!confirm('Envoyer à ' + label + ' ?')) return;
  status.textContent = '⏳ Envoi...';
  try {
    const r = await fetch('/admin/api/admin/broadcast', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message: msg, target, channelId }),
    });
    const data = await r.json();
    if (data.ok) {
      status.textContent = target === 'channel' ? '✅ Envoyé au canal ' + channelId : '✅ Envoyé à ' + data.sent + ' users (' + data.failed + ' échecs)';
      document.getElementById('broadcastMsg').value = '';
    } else status.textContent = '❌ ' + (data.error || 'Erreur');
  } catch (e) { status.textContent = '❌ ' + e.message; }
}

async function renderBans() {

async function unbanUser(id) {
  if (!confirm('Débannir cet utilisateur ?')) return;
  await apiSend('user', 'PATCH', { telegram_id: id, is_banned: false });
  showToast('Débanni');
  const users = await apiGet('users'); DATA.users = users.data || [];
  DATA.admins = DATA.users.filter(u => u.is_admin);
  renderUsers(); renderBans();
}

async function loadMarketplaceAdmin() {
  const list = document.getElementById('mpList');
  if (!list) return;
  list.innerHTML = '<div class="empty">Chargement...</div>';
  try {
    const r = await fetch('/admin/api/admin/marketplace?status=' + MP_FILTER);
    const d = await r.json();
    if (!d.ok) { list.innerHTML = emptyBox('Erreur : ' + (d.error || '?')); return; }
    MP_ADS = d.ads || [];
    MP_REPORTS = d.reports || [];
    MP_STATS = d.stats || {};
    const pendingCount = MP_STATS.totalPending || 0;
    const badge = document.getElementById('mpPendingBadge');
    if (badge) badge.textContent = pendingCount + ' en attente';
    renderMpStats();
    renderMpAds();
    renderMpReports();
  } catch (e) {
    list.innerHTML = emptyBox('Erreur : ' + e.message);
  }
}

function renderMpStats() {
  const list = document.getElementById('mpList');
  if (!list) return;
  const s = MP_STATS;
  list.innerHTML = '<div class="stats-grid" style="margin-bottom:16px;">' +
    '<div class="stat-box"><div class="lbl">Actives</div><div class="val">' + (s.totalActive || 0) + '</div></div>' +
    '<div class="stat-box" style="border-color:var(--kobo-orange);"><div class="lbl">En attente</div><div class="val">' + (s.totalPending || 0) + '</div></div>' +
    '<div class="stat-box" style="border-color:#3b82f6;"><div class="lbl">Ventes</div><div class="val">' + (s.totalOrders || 0) + '</div></div>' +
    '<div class="stat-box" style="border-color:var(--kobo-green);"><div class="lbl">Commissions</div><div class="val">' + Math.round(s.totalCommission || 0) + ' K</div></div>' +
  '</div>';
}

function renderMpAds() {
  const list = document.getElementById('mpList');
  if (!list) return;
  const q = (document.getElementById('mpSearch')?.value || '').toLowerCase().trim();
  let f = MP_ADS;
  if (q) f = f.filter(a => (a.title || '').toLowerCase().includes(q) || (a.description || '').toLowerCase().includes(q) || (a.category || '').toLowerCase().includes(q));
  let html = '';
  if (f.length === 0) html = emptyBox('Aucune annonce');
  else html = f.map(a => {
    const seller = a.users?.first_name || a.users?.username || 'Vendeur';
    const sellerId = a.users?.telegram_id || a.seller_id;
    const statusMap = { pending: { l: '⏳ En attente', c: 'orange' }, active: { l: '✅ Active', c: 'green' }, rejected: { l: '❌ Rejetée', c: 'red' } };
    const s = statusMap[a.status] || { l: a.status, c: 'gray' };
    const boosted = a.boosted && a.boosted_until && new Date(a.boosted_until) > new Date();
    return '<div class="item-card">' +
      '<div class="ic-img">' + (boosted ? '🚀' : '🛒') + '</div>' +
      '<div class="ic-body">' +
        '<div class="ic-name">' + escapeHtml(a.title) + ' <span class="badge ' + s.c + '">' + s.l + '</span>' + (boosted ? ' <span class="badge gold">🚀</span>' : '') + (a.broadcast_to_channels ? ' <span class="badge blue">📢</span>' : '') + '</div>' +
        '<div class="ic-meta"><span>💰 ' + Math.round(a.price) + ' ' + (a.currency === 'real' ? 'FCFA' : 'Kobo') + '</span><span class="dot">•</span><span>' + escapeHtml(a.category) + '</span></div>' +
        '<div class="ic-meta"><span>👤 ' + escapeHtml(seller) + '</span><span class="dot">•</span><span>ID ' + sellerId + '</span></div>' +
        '<div class="ic-meta" style="font-size:10.5px;"><span>' + timeAgo(a.created_at) + '</span><span class="dot">•</span><span>' + a.sold + '/' + (a.places || '∞') + '</span></div>' +
        (a.description ? '<div class="ic-desc">' + escapeHtml(a.description.slice(0, 150)) + (a.description.length > 150 ? '...' : '') + '</div>' : '') +
      '</div>' +
      '<div class="ic-actions">' +
        (a.status === 'pending' ?
          '<button class="icon-btn" onclick="mpApprove(' + a.id + ')">✅</button>' +
          '<button class="icon-btn" onclick="mpReject(' + a.id + ')">❌</button>'
          :
          (a.status === 'active' ? (boosted ? '<button class="icon-btn" onclick="mpUnboost(' + a.id + ')">⬇️</button>' : '<button class="icon-btn" onclick="mpBoost(' + a.id + ')">🚀</button>') : '')
        ) +
        '<button class="icon-btn" onclick="mpDelete(' + a.id + ')">🗑️</button>' +
      '</div>' +
    '</div>';
  }).join('');
  list.innerHTML += html;
}

function renderMpReports() {
  if (!MP_REPORTS || MP_REPORTS.length === 0) return;
  const list = document.getElementById('mpList');
  if (!list) return;
  list.innerHTML += '<h3 style="font-weight:800;color:#b91c1c;margin:20px 0 10px;">🚨 Signalements (' + MP_REPORTS.length + ')</h3>' +
    MP_REPORTS.map(r => '<div class="item-card" style="border-left:4px solid #dc2626;"><div class="ic-img">⚠️</div><div class="ic-body"><div class="ic-name">' + escapeHtml(r.marketplace_ads?.title || 'Annonce supprimée') + '</div><div class="ic-meta"><span>👤 ' + escapeHtml(r.users?.first_name || r.users?.username || 'User') + '</span></div><div class="ic-desc">🚩 ' + escapeHtml(r.reason) + '</div></div></div>').join('');
}

async function mpApprove(adId) {
  if (!confirm('Approuver ?')) return;
  const r = await fetch('/admin/api/admin/marketplace', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adId, action: 'approve' }) });
  const d = await r.json();
  if (d.error) return showToast('Erreur : ' + d.error, true);
  showToast('✅ Approuvée'); loadMarketplaceAdmin();
}
async function mpReject(adId) {
  const reason = prompt('Motif :');
  if (!reason) return;
  await fetch('/admin/api/admin/marketplace', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adId, action: 'reject', reason }) });
  showToast('❌ Rejetée'); loadMarketplaceAdmin();
}
async function mpBoost(adId) {
  await fetch('/admin/api/admin/marketplace', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adId, action: 'boost' }) });
  showToast('🚀 Boostée'); loadMarketplaceAdmin();
}
async function mpUnboost(adId) {
  await fetch('/admin/api/admin/marketplace', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adId, action: 'unboost' }) });
  showToast('Retiré'); loadMarketplaceAdmin();
}
async function mpDelete(adId) {
  if (!confirm('Supprimer ?')) return;
  await fetch('/admin/api/admin/marketplace', { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adId, action: 'delete' }) });
  showToast('Supprimée'); loadMarketplaceAdmin();
}

document.getElementById('mpFilters')?.addEventListener('click', (e) => {
  const c = e.target.closest('.filter-chip'); if (!c) return;
  document.querySelectorAll('#mpFilters .filter-chip').forEach(x => x.classList.remove('active'));
  c.classList.add('active');
  MP_FILTER = c.dataset.filter;
  loadMarketplaceAdmin();
});
